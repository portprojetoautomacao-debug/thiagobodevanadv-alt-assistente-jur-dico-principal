#!/bin/bash
# Corrigir todos os shell scripts

  # 1. Substituir [ por [[
  sed -i 's/if \[ /if [[ /g' "$script"
  sed -i 's/ \] ;/ ]] ;/g' "$script"

done
EOF
chmod +x fix-shell-issues.sh
